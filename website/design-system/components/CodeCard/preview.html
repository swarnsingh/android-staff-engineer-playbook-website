ProductRoute.ktCollect state at the screen boundary; children get values and lambdas 
@Composable
fun ProductRoute(viewModel: ProductViewModel) {
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()
    ProductScreen(products = uiState.products, onProductClick = viewModel::openProduct)
}

@Composable
fun ProductScreen(products: List<Product>, onProductClick: (ProductId) -> Unit) {
    LazyColumn {
        items(products, key = { product -> product.id }) { product ->
            ProductRow(product, onClick = { onProductClick(product.id) })
        }
    }
}
