@Composable
fun ProductRoute(viewModel: ProductViewModel) {
    val uiState by viewModel.uiState.collectAsStateWithLifecycle()
    // Children get plain values and lambdas, not the ViewModel
    ProductScreen(products = uiState.products, onProductClick = viewModel::openProduct)
    LazyColumn { items(uiState.products, key = { product -> product.id }) { /* … */ } }
}
